    45→    45→function phaseKeyOf(p: number): string {
    46→    46→  const b = [
    47→    47→    [0, 'phase.new'], [0.0625, 'phase.waxingCrescent'], [0.1875, 'phase.firstQuarter'],
    48→    48→    [0.3125, 'phase.waxingGibbous'], [0.4375, 'phase.full'], [0.5625, 'phase.waningGibbous'],
    49→    49→    [0.6875, 'phase.lastQuarter'], [0.8125, 'phase.waningCrescent'],
    50→    50→  ] as [number, string][];
    51→    51→  for (let i = b.length - 1; i >= 0; i--) {
    52→    52→    if (p >= b[i][0]) return b[i][1];
    53→    53→  }
    54→    54→  return 'phase.new';
    55→    55→}

    57→    57→/**
    58→    58→ * Live "right now" strip for the Home hero: current Choghadiya slot with a
    59→    59→ * ticking countdown + progress, and a Rahu-Kaal warning while it is active.
    60→    60→ * Ticks locally (1s) without re-rendering the parent screen.
    61→    61→ */
    62→    62→function LiveNowStrip({ choghadiya, rahuKaal, tz, locale }: {
    63→    63→  choghadiya: NonNullable<TodayPanchang['choghadiya']>;
    64→    64→  rahuKaal: TodayPanchang['rahuKaal'];
    65→    65→  tz: string;
    66→    66→  locale: string;
    67→    67→}) {
    68→    68→  const { t } = useTranslation();
    69→    69→  const [, setTick] = useState(0);
    70→    70→  useEffect(() => {
    71→    71→    const id = setInterval(() => setTick((n) => n + 1), 1000);
    72→    72→    return () => clearInterval(id);
    73→    73→  }, []);
    74→    74→
    75→    75→  const now = Date.now();
    76→    76→  const slots = [...choghadiya.day, ...choghadiya.night];
    77→    77→  const cur = slots.find((s) => new Date(s.startISO).getTime() <= now && new Date(s.endISO).getTime() > now);
    78→    78→  const rahuStart = rahuKaal?.startISO ? new Date(rahuKaal.startISO).getTime() : null;
    79→    79→  const rahuEnd = rahuKaal?.endISO ? new Date(rahuKaal.endISO).getTime() : null;
    80→    80→  const rahuActive = rahuStart != null && rahuEnd != null && rahuStart <= now && rahuEnd > now;
    81→    81→  if (!cur && !rahuActive) return null;
    82→    82→
    83→    83→  const start = cur ? new Date(cur.startISO).getTime() : 0;
    84→    84→  const end = cur ? new Date(cur.endISO).getTime() : 0;
    85→    85→  const pct = cur ? Math.min(100, Math.max(0, ((now - start) / (end - start)) * 100)) : 0;
    86→    86→  const remMs = Math.max(0, end - now);
    87→    87→  const remStr = remMs >= 3600_000
    88→    88→    ? `${Math.floor(remMs / 3600_000)}h ${Math.floor((remMs % 3600_000) / 60_000)}m`
    89→    89→    : `${Math.floor(remMs / 60_000)}m ${String(Math.floor((remMs % 60_000) / 1000)).padStart(2, '0')}s`;
    90→    90→  const fmtT = (iso: string) => new Date(iso).toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit', timeZone: tz });
    91→    91→
    92→    92→  return (
    93→    93→    <div className="mt-3.5 pt-3 border-t border-gold/15" data-testid="live-now-strip">
    94→    94→      {cur && (
    95→    95→        <div className="flex items-center gap-2.5 flex-wrap">
    96→    96→          <span className="inline-flex items-center gap-1.5 shrink-0">
    97→    97→            <span className="relative flex h-2 w-2">
    98→    98→              <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-gold opacity-60" />
    99→    99→              <span className="relative inline-flex rounded-full h-2 w-2 bg-gold" />
   100→   100→            </span>
   101→   101→            <span className="text-t12 font-bold uppercase tracking-wider text-gold">{t('home.liveTitle')}</span>
   102→   102→          </span>
   103→   103→          <span className="font-serif text-t16 font-semibold text-charcoal">{choghName(t, cur.name)}</span>
   104→   104→          <span
   105→   105→            className={cn(
   106→   106→              'rounded-full px-2 py-0.5 text-t12 font-semibold',
   107→   107→              cur.quality === 'good' ? 'bg-gold-faint text-[#7a8b6f]' : cur.quality === 'medium' ? 'bg-white/60 text-muted-foreground' : 'bg-maroon-soft text-maroon',
   108→   108→            )}
   109→   109→          >
   110→   110→            {t(`choghadiya.${cur.quality}`)}
   111→   111→          </span>
   112→   112→          <span className="text-t12 text-muted-foreground tabular-nums">{t('home.liveEndsIn', { time: remStr })}</span>
   113→   113→        </div>
   114→   114→      )}
   115→   115→      {cur && (
   116→   116→        <div className="mt-2 h-1 rounded-full bg-gold/15 overflow-hidden" role="progressbar" aria-valuenow={Math.round(pct)} aria-valuemin={0} aria-valuemax={100}>
   117→   117→          <div className="h-full gold-grad rounded-full" style={{ width: `${pct}%`, transition: 'width 1s linear' }} />
   118→   118→        </div>
   119→   119→      )}
   120→   120→      {rahuActive && rahuEnd && (
   121→   121→        <p className={cn('inline-flex items-center gap-1.5 rounded-full px-2.5 py-1 text-t12 font-semibold bg-maroon-soft text-maroon border border-maroon/25', cur && 'mt-2')}>
   122→   122→          <Icon name="warning" size={13} />
   123→   123→          {t('home.liveRahu', { time: fmtT(rahuKaal?.endISO as string) })}
   124→   124→        </p>
   125→   125→      )}
   126→   126→    </div>
   127→   127→  );
   128→   128→}
