    45→function phaseKeyOf(p: number): string {
    46→  const b = [
    47→    [0, 'phase.new'], [0.0625, 'phase.waxingCrescent'], [0.1875, 'phase.firstQuarter'],
    48→    [0.3125, 'phase.waxingGibbous'], [0.4375, 'phase.full'], [0.5625, 'phase.waningGibbous'],
    49→    [0.6875, 'phase.lastQuarter'], [0.8125, 'phase.waningCrescent'],
    50→  ] as [number, string][];
    51→  for (let i = b.length - 1; i >= 0; i--) {
    52→    if (p >= b[i][0]) return b[i][1];
    53→  }
    54→  return 'phase.new';
    55→}

    57→/**
    58→ * Live "right now" strip for the Home hero: current Choghadiya slot with a
    59→ * ticking countdown + progress, and a Rahu-Kaal warning while it is active.
    60→ * Ticks locally (1s) without re-rendering the parent screen.
    61→ */
    62→function LiveNowStrip({ choghadiya, rahuKaal, tz, locale }: {
    63→  choghadiya: NonNullable<TodayPanchang['choghadiya']>;
    64→  rahuKaal: TodayPanchang['rahuKaal'];
    65→  tz: string;
    66→  locale: string;
    67→}) {
    68→  const { t } = useTranslation();
    69→  const [, setTick] = useState(0);
    70→  useEffect(() => {
    71→    const id = setInterval(() => setTick((n) => n + 1), 1000);
    72→    return () => clearInterval(id);
    73→  }, []);
    74→
    75→  const now = Date.now();
    76→  const slots = [...choghadiya.day, ...choghadiya.night];
    77→  const cur = slots.find((s) => new Date(s.startISO).getTime() <= now && new Date(s.endISO).getTime() > now);
    78→  const rahuStart = rahuKaal?.startISO ? new Date(rahuKaal.startISO).getTime() : null;
    79→  const rahuEnd = rahuKaal?.endISO ? new Date(rahuKaal.endISO).getTime() : null;
    80→  const rahuActive = rahuStart != null && rahuEnd != null && rahuStart <= now && rahuEnd > now;
    81→  if (!cur && !rahuActive) return null;
    82→
    83→  const start = cur ? new Date(cur.startISO).getTime() : 0;
    84→  const end = cur ? new Date(cur.endISO).getTime() : 0;
    85→  const pct = cur ? Math.min(100, Math.max(0, ((now - start) / (end - start)) * 100)) : 0;
    86→  const remMs = Math.max(0, end - now);
    87→  const remStr = remMs >= 3600_000
    88→    ? `${Math.floor(remMs / 3600_000)}h ${Math.floor((remMs % 3600_000) / 60_000)}m`
    89→    : `${Math.floor(remMs / 60_000)}m ${String(Math.floor((remMs % 60_000) / 1000)).padStart(2, '0')}s`;
    90→  const fmtT = (iso: string) => new Date(iso).toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit', timeZone: tz });
    91→
    92→  return (
    93→    <div className="mt-3.5 pt-3 border-t border-gold/15" data-testid="live-now-strip">
    94→      {cur && (
    95→        <div className="flex items-center gap-2.5 flex-wrap">
    96→          <span className="inline-flex items-center gap-1.5 shrink-0">
    97→            <span className="relative flex h-2 w-2">
    98→              <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-gold opacity-60" />
    99→              <span className="relative inline-flex rounded-full h-2 w-2 bg-gold" />
   100→            </span>
   101→            <span className="text-t12 font-bold uppercase tracking-wider text-gold">{t('home.liveTitle')}</span>
   102→          </span>
   103→          <span className="font-serif text-t16 font-semibold text-charcoal">{choghName(t, cur.name)}</span>
   104→          <span
   105→            className={cn(
   106→              'rounded-full px-2 py-0.5 text-t12 font-semibold',
   107→              cur.quality === 'good' ? 'bg-gold-faint text-[#7a8b6f]' : cur.quality === 'medium' ? 'bg-white/60 text-muted-foreground' : 'bg-maroon-soft text-maroon',
   108→            )}
   109→          >
   110→            {t(`choghadiya.${cur.quality}`)}
   111→          </span>
   112→          <span className="text-t12 text-muted-foreground tabular-nums">{t('home.liveEndsIn', { time: remStr })}</span>
   113→        </div>
   114→      )}
   115→      {cur && (
   116→        <div className="mt-2 h-1 rounded-full bg-gold/15 overflow-hidden" role="progressbar" aria-valuenow={Math.round(pct)} aria-valuemin={0} aria-valuemax={100}>
   117→          <div className="h-full gold-grad rounded-full" style={{ width: `${pct}%`, transition: 'width 1s linear' }} />
   118→        </div>
   119→      )}
   120→      {rahuActive && rahuEnd && (
   121→        <p className={cn('inline-flex items-center gap-1.5 rounded-full px-2.5 py-1 text-t12 font-semibold bg-maroon-soft text-maroon border border-maroon/25', cur && 'mt-2')}>
   122→          <Icon name="warning" size={13} />
   123→          {t('home.liveRahu', { time: fmtT(rahuKaal?.endISO as string) })}
   124→        </p>
   125→      )}
   126→    </div>
   127→  );
   128→}

   465→
   466→      {/* Dasha transition countdown — ink card */}
   467→      {dashaNow && dashaNow.next.lord && (
   468→        <GlassCard className="p-5 sm:p-6 glass-ink relative overflow-hidden">
   469→          <div className="flex items-start justify-between gap-3">

   472→              <p className="mt-2 text-t16 font-serif text-gold-grad font-semibold">
   473→                {t('dasha.current', { maha: t(`planet.${dashaNow.mahaLord}`), antar: dashaNow.antarLord ? t(`planet.${dashaNow.antarLord}`) : '' })}
   474→              </p>
   475→              <p className="mt-1 text-t14 text-[#cdb98e]">
   476→                {dashaNow.next.daysLeft <= 0
   477→                  ? t('dasha.endsToday')
   478→                  : dashaNow.next.isMahaChange
   479→                    ? t('dasha.mahaChange', { next: t(`planet.${dashaNow.next.lord}`) })
   480→                    : t('dasha.endsIn', { n: dashaNow.next.daysLeft, next: t(`planet.${dashaNow.next.lord}`) })}
   481→                <span className="text-[#a5947a]"> · {new Date(`${dashaNow.next.dateISO}T12:00:00Z`).toLocaleDateString(locale, { day: 'numeric', month: 'short' })}</span>
   482→              </p>
   483→            </div>

   485→              type="button"
   486→              onClick={() => onNavigate('kundli', 'reports')}
   487→              className="shrink-0 text-t12 text-gold hover:text-gold-light underline underline-offset-4 decoration-gold/30 transition-colors duration-200"
   488→            >
   489→              {t('dasha.viewReports')}

   495→          </div>
   496→          <div className="mt-1.5 flex justify-between text-t12 text-[#a5947a]">
   497→            <span>{t(`planet.${dashaNow.mahaLord}`)}-{dashaNow.antarLord ? t(`planet.${dashaNow.antarLord}`) : ''}</span>
   498→            <span>{dashaNow.antarPercent}%</span>
   499→          </div>
   500→        </GlassCard>
   501→      )}
   502→
   503→      {/* Daily check-in */}
   504→      <GlassCard className="p-5 sm:p-6">
   505→        <AnimatePresence mode="wait">
   506→          {checkin?.checkedInToday ? (
   507→            <motion.div
   508→              key="done"
   509→              initial={{ opacity: 0, y: 8 }}
   510→              animate={{ opacity: 1, y: 0 }}
   511→              exit={{ opacity: 0, y: -8 }}
   512→              className="flex items-center gap-4"
   513→            >
   514→              <span className="shrink-0 w-12 h-12 rounded-full gold-grad flex items-center justify-center text-charcoal">
   515→                <Icon name="check" size={24} strokeWidth={2} />
   516→              </span>
   517→              <div className="flex-1 min-w-0">
   518→                <p className="text-t16 font-semibold text-charcoal">{t('home.checkinDone')}</p>
   519→                <p className="text-t12 text-muted-foreground">{t('home.checkinDoneBody')}</p>
   520→              </div>
   521→              <div className="text-right shrink-0">
   522→                <span className="inline-flex items-center gap-1 text-maroon font-semibold text-t16">
   523→                  <Icon name="flame" size={18} className="flame-pulse" />
   524→                  {checkin.streak}
   525→                </span>
   526→                <p className="text-t12 text-muted-foreground">{t('home.streak', { n: checkin.streak })}</p>

   538→            <motion.div key="form" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0, y: -8 }}>

   541→                <span className="inline-flex items-center gap-1 text-t12 text-muted-foreground">
   542→                  <Icon name="flame" size={14} className="text-gold" />
   543→                  {checkin.streak > 0 ? t('home.streak', { n: checkin.streak }) : t('home.streakNew')}

   545→              </div>
   546→
   547→              <div className="mt-4 space-y-4">
   548→                <MoodSlider
   549→                  label={t('home.mood')}

   551→                  onChange={setMood}
   552→                  lowLabel={language === 'hi' ? 'भारी' : 'Heavy'}

   555→                <MoodSlider
   556→                  label={t('home.energy')}
   557→                  value={energy}

   563→                  value={note}

   566→                  rows={2}
   567→                  className="resize-none rounded-xl border-gold/25 bg-white/70 text-t14 focus-visible:ring-gold/30 max-h-24"

   569→                />

   572→                  {saving ? t('common.loading') : justSaved ? t('common.saved') : t('home.checkinSubmit')}
   573→                  {!saving && <Icon name="check" size={16} />}
   574→                </GoldButton>
   575→              </div>
   576→            </motion.div>
   577→          )}
   578→        </AnimatePresence>
   579→      </GlassCard>
   580→
   581→      {/* Weekly reflection letter (ink card) */}
   582→      <div>
   583→        <SectionHeader title={t('home.weeklyTitle')} icon="journal" />
   584→        <GlassCard ink className="sheen-sweep p-5 sm:p-6">
   585→          {checkin?.reflection ? (
   586→            <>
   587→              <button type="button" onClick={() => setLetterOpen((v) => !v)} className="w-full text-left">
   588→                <div className="flex items-center justify-between gap-3">
   589→                  <div className="min-w-0">
   590→                    <p className="text-t12 uppercase tracking-wider text-gold font-semibold">{t('home.weeklyReady')}</p>
   591→                    <p className="font-serif text-t20 mt-1 text-[#f0e4c3] truncate">
   592→                      {checkin.reflection.reflectionText.split('\n')[0].slice(0, 80)}…
   593→                    </p>
   594→                    {weekRangeLabel && <p className="mt-1 text-t12 text-[#a5947a] italic">{weekRangeLabel}</p>}
   595→                  </div>
   596→                  <span className="shrink-0 flex flex-col items-center gap-1" aria-hidden>
   597→                    <span className="inline-flex items-center justify-center w-9 h-9 rounded-full border border-gold/40 bg-gold/10 rotate-45">
   598→                      <span className="block w-3.5 h-3.5 rounded-[2px] bg-gradient-to-br from-[#E8C568] to-[#C9A227] shadow-[0_0_10px_rgba(232,197,104,0.45)]" />

   600→                    <Icon name="chevron-down" size={16} className={cn('text-gold transition-transform duration-250', letterOpen && 'rotate-180')} />

   602→                </div>
   603→              </button>

   615→                    </p>
   616→                    <div className="ornament-divider mt-4" style={{ color: 'rgba(232,197,104,0.45)' }} aria-hidden>✦</div>
   617→                    <button
   618→                      type="button"
   619→                      onClick={toggleLetterAudio}
   620→                      className="mt-3 inline-flex items-center gap-2 rounded-full bg-gold/15 border border-gold/40 px-4 py-1.5 text-t12 font-medium text-gold-light hover:bg-gold/25 active:scale-95 transition-all duration-250"

   630→          ) : (
   631→            <div className="flex items-center gap-4">

   636→                <p className="text-t14 text-[#e8dcbf]">{t('home.weeklyEmpty')}</p>

   638→                  <div className="mt-2 h-1.5 rounded-full bg-white/10 overflow-hidden">
   639→                    <div
   640→                      className="h-full gold-grad rounded-full transition-[width] duration-500"
   641→                      style={{ width: `${Math.min(100, (checkin.weekEntryCount / 3) * 100)}%` }}

   643→                  </div>

   645→                {checkin && <p className="mt-1 text-t12 text-[#a5947a]">{t('home.weeklyProgress', { n: checkin.weekEntryCount })}</p>}

   649→        </GlassCard>
   650→      </div>
   651→
   652→      {/* Feature grid */}
   653→      {!chart && (
   654→        <GlassCard className="p-5 flex items-center gap-4 border-maroon/25">
   655→          <span className="inline-flex items-center justify-center w-11 h-11 rounded-full bg-maroon-soft text-maroon shrink-0">
   656→            <Icon name="warning" size={20} />

   666→      {features.sections.map((section, si) => (

   670→          <div className="grid grid-cols-2 lg:grid-cols-3 gap-3 sm:gap-4">

   673→                key={item.titleKey}
   674→                onClick={() => onNavigate('kundli', item.view)}
   675→                className="p-4 sm:p-5 group"

   677→                <span className="inline-flex items-center justify-center w-10 h-10 rounded-xl bg-gold-soft/80 text-gold transition-all duration-250 group-hover:gold-grad group-hover:text-charcoal">

   679→                </span>
   680→                <p className="mt-3 text-t14 font-semibold text-charcoal">{t(item.titleKey)}</p>

   683→            ))}
   684→          </div>

   686→      ))}
   687→    </div>
   688→  );
   689→}
