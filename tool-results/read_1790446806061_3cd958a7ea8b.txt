    67→/** Compose a human-readable classical timing from festival month+anchor (e.g. "Bhadrapada, Shukla 15"). */
    68→function anchorText(f: FestivalInfo): string | undefined {
    69→  if (!f.anchor) return undefined;
    70→  const paksha = f.anchor.startsWith('S') ? 'Shukla' : 'Krishna';
    71→  const num = f.anchor.slice(1);
    72→  const monthName = f.month ? f.month.charAt(0).toUpperCase() + f.month.slice(1) : '';
    73→  return [monthName, `${paksha} ${num}`].filter(Boolean).join(', ');
    74→}

   336→function shiftMonth(month: string, delta: number, set: (v: string) => void) {
   337→  const [y, m] = month.split('-').map(Number);
   338→  const d = new Date(Date.UTC(y, m - 1 + delta, 1));
   339→  set(`${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, '0')}`);
   340→}

   342→function monthLabel(month: string, locale: string): string {
   343→  const [y, m] = month.split('-').map(Number);
   344→  return new Date(Date.UTC(y, m - 1, 1)).toLocaleDateString(locale, { month: 'long', year: 'numeric', timeZone: 'UTC' });
   345→}

   349→function MonthGrid({
   350→  month,
   351→  days,
   352→  today,
   353→  locale,
   354→  language,
   355→  onPrev,
   356→  onNext,
   357→  onPickDay,
   358→  onInfo,
   359→  prevLabel,
   360→  nextLabel,
   361→}: {
   362→  month: string;
   363→  days: MonthDay[];
   364→  today: string;
   365→  locale: string;
   366→  language: 'en' | 'hi';
   367→  onPrev: () => void;
   368→  onNext: () => void;
   369→  onPickDay: (date: string) => void;
   370→  onInfo: (date: string, f: FestivalInfo) => void;
   371→  prevLabel: string;
   372→  nextLabel: string;
   373→}) {
   374→  const { t } = useTranslation();
   375→  const [y, m] = month.split('-').map(Number);
   376→  const firstWeekday = days[0]?.weekday ?? new Date(Date.UTC(y, m - 1, 1)).getUTCDay();
   377→
   378→  // Dominant purnimanta lunar month across the displayed days + adhika flag
   379→  // (a leap month repeats the previous name — any day marked adhika wins).
   380→  const lunarInfo = (() => {
   381→    const counts = new Map<string, { n: number; name: NonNullable<MonthDay['lunarMonth']> }>();
   382→    for (const d of days) {
   383→      if (!d.lunarMonth) continue;
   384→      const cur = counts.get(d.lunarMonth.key) ?? { n: 0, name: d.lunarMonth };
   385→      cur.n += 1;
   386→      if (d.lunarMonth.adhika) cur.name = d.lunarMonth;
   387→      counts.set(d.lunarMonth.key, cur);
   388→    }
   389→    const dominant = [...counts.values()].sort((a, b) => b.n - a.n)[0];
   390→    return dominant ? { name: dominant.name, adhika: dominant.name.adhika === true } : null;
   391→  })();
   392→
   393→  return (
   394→    <div>
   395→      {/* Month navigation */}
   396→      <div className="flex items-center justify-between gap-2 mb-4">
   397→        <button
   398→          type="button"
   399→          onClick={onPrev}
   400→          aria-label={prevLabel}
   401→          className="w-9 h-9 rounded-full glass inline-flex items-center justify-center text-gold hover:border-gold/50 active:scale-90 transition-all duration-200"
   402→        >
   403→          <Icon name="chevron-left" size={16} />
   404→        </button>
   405→        <p className="font-serif text-t20 text-charcoal">{monthLabel(month, locale)}</p>
   406→        <button
   407→          type="button"
   408→          onClick={onNext}
   409→          aria-label={nextLabel}
   410→          className="w-9 h-9 rounded-full glass inline-flex items-center justify-center text-gold hover:border-gold/50 active:scale-90 transition-all duration-200"
   411→        >
   412→          <Icon name="chevron-right" size={16} />
   413→        </button>
   414→      </div>
   415→
   416→      {/* Lunar month + adhika strip */}
   417→      {lunarInfo && (
   418→        <div className="flex items-center gap-2 mb-3 flex-wrap">
   419→          <span className="inline-flex items-center gap-1.5 rounded-full bg-white/60 border border-gold/25 px-2.5 py-1 text-t12 text-muted-foreground">
   420→            <Icon name="moon" size={13} className="text-gold" />
   421→            {t('panchang.lunarMonth')}:{' '}
   422→            <span className="text-charcoal font-medium">{language === 'hi' ? lunarInfo.name.hi : lunarInfo.name.en}</span>
   423→          </span>
   424→          {lunarInfo.adhika && (
   425→            <span className="inline-flex items-center rounded-full bg-gold-faint/80 border border-gold/40 px-2.5 py-1 text-t12 font-semibold text-[#8a6d1f]">
   426→              {t('panchang.adhika')}
   427→            </span>
   428→          )}
   429→        </div>
   430→      )}
   431→
   432→      {/* Weekday header */}
   433→      <div className="grid grid-cols-7 gap-1 mb-1.5">
   434→        {WEEKDAY_MIN.map((w, i) => (
   435→          <div key={w} className={cn('text-center text-t12 font-semibold', i === 0 || i === 6 ? 'text-maroon/70' : 'text-muted-foreground')}>
   436→            {t(`weekdaysShort.${i}`)}
   437→          </div>
   438→        ))}
   439→      </div>

   611→/**
   612→ * Festival / parva significance dialog — Acharya-generated classical note,
   613→ * cached per (lang|kind|name) in localStorage so it generates only once.
   614→ */
   615→function FestivalDetailDialog({ detail, onClose }: { detail: FestDetail | null; onClose: () => void }) {
   616→  const { t, i18n } = useTranslation();
   617→  const language = useAppStore((s) => s.language);
   618→  const locale = language === 'hi' ? 'hi-IN' : 'en-IN';
   619→  const [text, setText] = useState<string | null>(null);
   620→  const [state, setState] = useState<'loading' | 'ready' | 'error'>('loading');
   621→  // spoken version of the note — cached in localStorage (quota-guarded)
   622→  const [audioState, setAudioState] = useState<'idle' | 'loading' | 'playing'>('idle');
   623→  const audioRef = useRef<HTMLAudioElement | null>(null);
   624→
   625→  const audioCacheKey = detail ? `ss-fest-audio-${language}-${detail.kind}-${detail.en}` : null;
   626→
   627→  const stopAudio = useCallback(() => {
   628→    audioRef.current?.pause();
   629→    audioRef.current = null;
   630→    setAudioState('idle');
   631→  }, []);
   632→
   633→  // The dialog is remounted per detail (keyed by parent), so audio state and
   634→  // playback reset naturally on change; stop on unmount only.
   635→  useEffect(() => () => stopAudio(), [stopAudio]);
   636→
   637→  const toggleAudio = async () => {
   638→    if (audioState === 'playing') {
   639→      stopAudio();
   640→      return;
   641→    }
   642→    if (audioRef.current) {
   643→      void audioRef.current.play().catch(() => setAudioState('idle'));
   644→      setAudioState('playing');
   645→      return;
   646→    }
   647→    if (!text || !audioCacheKey) return;
   648→    setAudioState('loading');
   649→    try {
   650→      let uri: string | null = null;
   651→      try {
   652→        uri = localStorage.getItem(audioCacheKey);
   653→      } catch {
   654→        /* private mode */
   655→      }
   656→      if (!uri) {
   657→        const res = await api<{ audioDataUri: string }>('/api/festival-audio', {
   658→          method: 'POST',
   659→          body: JSON.stringify({ kind: detail?.kind, name: detail?.en, language, text }),
   660→        });
   661→        uri = res.audioDataUri;
   662→        try {
   663→          localStorage.setItem(audioCacheKey, uri);
   664→        } catch {
   665→          /* quota — skip caching */
   666→        }
   667→      }
   668→      const audio = new Audio(uri);
   669→      audio.onended = () => {
   670→        setAudioState('idle');
   671→        audioRef.current = null;
   672→      };
   673→      audioRef.current = audio;
   674→      void audio.play().catch(() => setAudioState('idle'));
   675→      setAudioState('playing');
   676→    } catch {
   677→      setAudioState('idle');
   678→    }
   679→  };
   680→
   681→  useEffect(() => {
   682→    if (!detail) return;
   683→    let cancelled = false;
   684→    const run = async () => {
   685→      setState('loading');
   686→      setText(null);
   687→      const cacheKey = `ss-fest-${language}-${detail.kind}-${detail.en}`;
   688→      try {
   689→        const hit = localStorage.getItem(cacheKey);
   690→        if (hit) {
   691→          if (!cancelled) {
   692→            setText(hit);
   693→            setState('ready');
   694→          }
   695→          return;
   696→        }
   697→      } catch {
   698→        /* private mode — skip cache */
   699→      }
   700→      try {
   701→        const res = await api<{ text: string }>('/api/festival-detail', {
   702→          method: 'POST',
   703→          body: JSON.stringify({ name: detail.en, kind: detail.kind, language, anchor: detail.anchor }),
   704→        });
   705→        if (cancelled) return;
   706→        setText(res.text);
   707→        setState('ready');
   708→        try {
   709→          localStorage.setItem(cacheKey, res.text);
   710→        } catch {
   711→          /* ignore */
   712→        }
   713→      } catch {
   714→        if (!cancelled) setState('error');
   715→      }
   716→    };
   717→    void run();
   718→    return () => {
   719→      cancelled = true;
   720→    };
   721→  }, [detail, language]);
   722→
   723→  const kindLabel = detail ? t(`panchang.festKind.${detail.kind}`) : '';
   724→  const dateLine = detail
   725→    ? new Date(`${detail.date}T12:00:00Z`).toLocaleDateString(locale, { weekday: 'long', day: 'numeric', month: 'long', timeZone: 'UTC' })
   726→    : '';
   727→
   728→  return (
   729→    <Dialog open={!!detail} onOpenChange={(v) => !v && onClose()}>
   730→      <DialogContent className="glass border-gold/30 max-w-md p-0 overflow-hidden gap-0" showCloseButton>
   731→        {detail && (
   732→          <>
   733→            <DialogHeader className="px-5 pt-5 pb-3 border-b border-gold/15 bg-gold-faint/50">
   734→              <div className="flex items-center gap-2">
   735→                <span
   736→                  className={cn(
   737→                    'rounded-full px-2 py-0.5 text-[10px] font-semibold uppercase tracking-wide',
   738→                    detail.kind === 'sankranti' ? 'bg-sage/15 text-[#5c6b52]' : detail.kind === 'ekadashi' ? 'bg-gold-faint text-[#8a6d1f]' : 'bg-gold-soft text-[#7a5c10]',
   739→                  )}
   740→                >
   741→                  {kindLabel}
   742→                </span>
   743→                <DialogDescription className="text-t12 text-muted-foreground">{dateLine}</DialogDescription>
   744→              </div>
   745→              <DialogTitle className="font-serif text-t20 text-charcoal mt-1.5">{language === 'hi' ? detail.hi : detail.en}</DialogTitle>
   746→            </DialogHeader>
   747→            <div className="px-5 py-4">
   748→              {state === 'loading' && (
   749→                <div className="space-y-2.5 py-1" aria-busy="true">
   750→                  <div className="h-3.5 w-full rounded shimmer" />
   751→                  <div className="h-3.5 w-11/12 rounded shimmer" />
   752→                  <div className="h-3.5 w-4/5 rounded shimmer" />
   753→                  <div className="h-3.5 w-3/5 rounded shimmer" />
   754→                  <p className="text-t12 text-muted-foreground pt-1">{t('panchang.festLoading')}</p>
   755→                </div>
   756→              )}
   757→              {state === 'error' && (
   758→                <ErrorState message={t('common.networkError')} />
   759→              )}
   760→              {state === 'ready' && text && (
   761→                <div className="rounded-xl glass-ink px-4 py-3.5 relative overflow-hidden">
   762→                  <div className="flex items-center gap-2.5 mb-2.5">
   763→                    <AcharyaAvatar size={30} />
   764→                    <div>
   765→                      <p className="text-t12 font-semibold text-gold-grad font-serif">Acharya Abhishek Sharma</p>
   766→                      <p className="text-[10px] text-[#a5947a]">{t('panchang.festBy')}</p>
   767→                    </div>
   768→                  </div>
   769→                  <p className="text-t14 leading-relaxed text-[#e8ddc8] whitespace-pre-line">{text}</p>
   770→                </div>
   771→              )}
   772→              {state === 'ready' && text && (
   773→                <button
   774→                  type="button"
   775→                  onClick={() => void toggleAudio()}
   776→                  disabled={audioState === 'loading'}
   777→                  title={t('panchang.festListen')}
   778→                  className="mt-3 inline-flex items-center gap-1.5 rounded-full border border-gold/30 bg-gold-faint/50 px-3 py-1.5 text-t12 text-gold hover:bg-gold-faint hover:border-gold/50 active:scale-95 transition-all duration-250 disabled:opacity-60"

   782→                </button>
   783→              )}
   784→            </div>
   785→          </>
   786→        )}
   787→      </DialogContent>
   788→    </Dialog>
   789→  );
   790→}
