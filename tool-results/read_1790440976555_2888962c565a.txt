    40→function phaseKeyOf(p: number): string {
    41→  const b = [
    42→    [0, 'phase.new'], [0.0625, 'phase.waxingCrescent'], [0.1875, 'phase.firstQuarter'],
    43→    [0.3125, 'phase.waxingGibbous'], [0.4375, 'phase.full'], [0.5625, 'phase.waningGibbous'],
    44→    [0.6875, 'phase.lastQuarter'], [0.8125, 'phase.waningCrescent'],
    45→  ] as [number, string][];
    46→  for (let i = b.length - 1; i >= 0; i--) {
    47→    if (p >= b[i][0]) return b[i][1];
    48→  }
    49→  return 'phase.new';
    50→}

   443→                  className="resize-none rounded-xl border-gold/25 bg-white/70 text-t14 focus-visible:ring-gold/30 max-h-24"
   444→                  maxLength={280}
   445→                />
   446→                {mood <= 2 && <p className="text-t12 text-maroon/80 italic">{t('home.checkinLow')}</p>}
   447→                <GoldButton onClick={submitCheckin} disabled={saving} className="w-full">
   448→                  {saving ? t('common.loading') : justSaved ? t('common.saved') : t('home.checkinSubmit')}
   449→                  {!saving && <Icon name="check" size={16} />}
   450→                </GoldButton>
   451→              </div>
   452→            </motion.div>
   453→          )}
   454→        </AnimatePresence>
   455→      </GlassCard>
   456→
   457→      {/* Weekly reflection letter (ink card) */}
   458→      <div>
   459→        <SectionHeader title={t('home.weeklyTitle')} icon="journal" />
   460→        <GlassCard ink className="sheen-sweep p-5 sm:p-6">
   461→          {checkin?.reflection ? (
   462→            <>
   463→              <button type="button" onClick={() => setLetterOpen((v) => !v)} className="w-full text-left">
   464→                <div className="flex items-center justify-between gap-3">
   465→                  <div className="min-w-0">
   466→                    <p className="text-t12 uppercase tracking-wider text-gold font-semibold">{t('home.weeklyReady')}</p>
   467→                    <p className="font-serif text-t20 mt-1 text-[#f0e4c3] truncate">
   468→                      {checkin.reflection.reflectionText.split('\n')[0].slice(0, 80)}…
   469→                    </p>
   470→                  </div>
   471→                  <Icon name="chevron-down" size={20} className={cn('text-gold transition-transform duration-250 shrink-0', letterOpen && 'rotate-180')} />
   472→                </div>
   473→              </button>
   474→              <AnimatePresence>
   475→                {letterOpen && (
   476→                  <motion.div
   477→                    initial={{ opacity: 0, height: 0 }}
   478→                    animate={{ opacity: 1, height: 'auto' }}
   479→                    exit={{ opacity: 0, height: 0 }}
   480→                    transition={{ duration: 0.3, ease: 'easeOut' }}
   481→                    className="overflow-hidden"
   482→                  >
   483→                    <p className="mt-4 whitespace-pre-wrap text-t14 leading-relaxed text-[#e8dcbf] letter-body">
   484→                      {checkin.reflection.reflectionText}
   485→                    </p>
   486→                    <button
   487→                      type="button"
   488→                      onClick={toggleLetterAudio}
   489→                      className="mt-4 inline-flex items-center gap-2 rounded-full bg-gold/15 border border-gold/40 px-4 py-1.5 text-t12 font-medium text-gold-light hover:bg-gold/25 active:scale-95 transition-all duration-250"
   490→                    >
   491→                      <Icon name={audioState === 'playing' ? 'pause' : 'speaker'} size={14} />
   492→                      {audioState === 'loading' ? t('letter.loadingAudio') : audioState === 'playing' ? t('letter.stop') : t('letter.listen')}
   493→                      {audioState === 'loading' && <span className="w-2 h-2 rounded-full bg-gold animate-pulse" />}
   494→                    </button>
   495→                  </motion.div>
   496→                )}
   497→              </AnimatePresence>
   498→            </>
   499→          ) : (
   500→            <div className="flex items-center gap-4">
   501→              <span className="shrink-0 inline-flex items-center justify-center w-12 h-12 rounded-full bg-gold/15 text-gold">
   502→                <Icon name="journal" size={22} />
   503→              </span>
   504→              <div className="flex-1 min-w-0">
   505→                <p className="text-t14 text-[#e8dcbf]">{t('home.weeklyEmpty')}</p>
   506→                {checkin && (
   507→                  <div className="mt-2 h-1.5 rounded-full bg-white/10 overflow-hidden">
   508→                    <div
   509→                      className="h-full gold-grad rounded-full transition-[width] duration-500"
   510→                      style={{ width: `${Math.min(100, (checkin.weekEntryCount / 3) * 100)}%` }}
   511→                    />
   512→                  </div>
   513→                )}
   514→                {checkin && <p className="mt-1 text-t12 text-[#a5947a]">{t('home.weeklyProgress', { n: checkin.weekEntryCount })}</p>}
   515→              </div>
   516→            </div>
   517→          )}
   518→        </GlassCard>
   519→      </div>
   520→
   521→      {/* Feature grid */}
   522→      {!chart && (
   523→        <GlassCard className="p-5 flex items-center gap-4 border-maroon/25">
   524→          <span className="inline-flex items-center justify-center w-11 h-11 rounded-full bg-maroon-soft text-maroon shrink-0">
   525→            <Icon name="warning" size={20} />
   526→          </span>
   527→          <div className="flex-1 min-w-0">
   528→            <p className="text-t14 font-semibold text-charcoal">{t('home.noChart')}</p>
   529→            <p className="text-t12 text-muted-foreground">{t('home.noChartBody')}</p>
   530→          </div>
   531→          <GoldButton onClick={() => onNavigate('profile')} className="shrink-0 h-9 px-4 text-t12">{t('profile.addChart')}</GoldButton>
   532→        </GlassCard>
   533→      )}
   534→
   535→      {features.sections.map((section, si) => (
   536→        <section key={section.titleKey} aria-label={t(section.titleKey)}>
   537→          {si > 0 && <div className="ornament-divider my-6">✦</div>}
   538→          <SectionHeader title={t(section.titleKey)} />
   539→          <div className="grid grid-cols-2 lg:grid-cols-3 gap-3 sm:gap-4">
   540→            {section.items.map((item) => (
   541→              <GlassCard
   542→                key={item.titleKey}
   543→                onClick={() => onNavigate('kundli', item.view)}
   544→                className="p-4 sm:p-5 group"
   545→              >
   546→                <span className="inline-flex items-center justify-center w-10 h-10 rounded-xl bg-gold-soft/80 text-gold transition-all duration-250 group-hover:gold-grad group-hover:text-charcoal">
   547→                  <Icon name={item.icon} size={20} />
   548→                </span>
   549→                <p className="mt-3 text-t14 font-semibold text-charcoal">{t(item.titleKey)}</p>
   550→                <p className="text-t12 text-muted-foreground mt-0.5 leading-snug">{t(item.bodyKey)}</p>
   551→              </GlassCard>
   552→            ))}
   553→          </div>
   554→        </section>
   555→      ))}
   556→    </div>
   557→  );
   558→}
