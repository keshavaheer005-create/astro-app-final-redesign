   107→   107→            className={cn(
   108→   108→              'rounded-full px-2 py-0.5 text-t12 font-semibold',
   109→   109→              cur.quality === 'good' ? 'bg-gold-faint text-[#7a8b6f]' : cur.quality === 'medium' ? 'bg-white/60 text-muted-foreground' : 'bg-burgundy-soft text-burgundy',

   119→   119→          <div className="h-full gold-grad rounded-full" style={{ width: `${pct}%`, transition: 'width 1s linear' }} />
   120→   120→        </div>
   121→   121→      )}
   122→   122→      {rahuActive && rahuEnd && (
   123→   123→        <p className={cn('inline-flex items-center gap-1.5 rounded-full px-2.5 py-1 text-t12 font-semibold bg-burgundy-soft text-burgundy border border-burgundy/25', cur && 'mt-2')}>
