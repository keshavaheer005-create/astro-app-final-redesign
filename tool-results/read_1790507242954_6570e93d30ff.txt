    47→function phaseKeyOf(p: number): string {
    48→  const b = [
    49→    [0, 'phase.new'], [0.0625, 'phase.waxingCrescent'], [0.1875, 'phase.firstQuarter'],
    50→    [0.3125, 'phase.waxingGibbous'], [0.4375, 'phase.full'], [0.5625, 'phase.waningGibbous'],
    51→    [0.6875, 'phase.lastQuarter'], [0.8125, 'phase.waningCrescent'],
    52→  ] as [number, string][];
    53→  for (let i = b.length - 1; i >= 0; i--) {
    54→    if (p >= b[i][0]) return b[i][1];
    55→  }
    56→  return 'phase.new';
    57→}

    59→/**
    60→ * Live "right now" strip for the Home hero: current Choghadiya slot with a
    61→ * ticking countdown + progress, and a Rahu-Kaal warning while it is active.
    62→ * Ticks locally (1s) without re-rendering the parent screen.
    63→ */
    64→function LiveNowStrip({ choghadiya, rahuKaal, tz, locale }: {
    65→  choghadiya: NonNullable<TodayPanchang['choghadiya']>;
    66→  rahuKaal: TodayPanchang['rahuKaal'];
    67→  tz: string;
    68→  locale: string;
    69→}) {
    70→  const { t } = useTranslation();
    71→  const [, setTick] = useState(0);
    72→  useEffect(() => {
    73→    const id = setInterval(() => setTick((n) => n + 1), 1000);
    74→    return () => clearInterval(id);
    75→  }, []);
    76→
    77→  const now = Date.now();
    78→  const slots = [...choghadiya.day, ...choghadiya.night];
    79→  const cur = slots.find((s) => new Date(s.startISO).getTime() <= now && new Date(s.endISO).getTime() > now);
    80→  const rahuStart = rahuKaal?.startISO ? new Date(rahuKaal.startISO).getTime() : null;
    81→  const rahuEnd = rahuKaal?.endISO ? new Date(rahuKaal.endISO).getTime() : null;
    82→  const rahuActive = rahuStart != null && rahuEnd != null && rahuStart <= now && rahuEnd > now;
    83→  if (!cur && !rahuActive) return null;
    84→
    85→  const start = cur ? new Date(cur.startISO).getTime() : 0;
    86→  const end = cur ? new Date(cur.endISO).getTime() : 0;
    87→  const pct = cur ? Math.min(100, Math.max(0, ((now - start) / (end - start)) * 100)) : 0;
    88→  const remMs = Math.max(0, end - now);
    89→  const remStr = remMs >= 3600_000
    90→    ? `${Math.floor(remMs / 3600_000)}h ${Math.floor((remMs % 3600_000) / 60_000)}m`
    91→    : `${Math.floor(remMs / 60_000)}m ${String(Math.floor((remMs % 60_000) / 1000)).padStart(2, '0')}s`;
    92→  const fmtT = (iso: string) => new Date(iso).toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit', timeZone: tz });
    93→
    94→  return (
    95→    <div className="mt-3.5 pt-3 border-t border-gold/15" data-testid="live-now-strip">
    96→      {cur && (
    97→        <div className="flex items-center gap-2.5 flex-wrap">
    98→          <span className="inline-flex items-center gap-1.5 shrink-0">
    99→            <span className="relative flex h-2 w-2">
   100→              <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-gold opacity-60" />
   101→              <span className="relative inline-flex rounded-full h-2 w-2 bg-gold" />
   102→            </span>
   103→            <span className="text-t12 font-bold uppercase tracking-wider text-gold">{t('home.liveTitle')}</span>
   104→          </span>
   105→          <span className="font-serif text-t16 font-semibold text-charcoal">{choghName(t, cur.name)}</span>
   106→          <span
   107→            className={cn(
   108→              'rounded-full px-2 py-0.5 text-t12 font-semibold',
   109→              cur.quality === 'good' ? 'bg-gold-faint text-[#7a8b6f]' : cur.quality === 'medium' ? 'bg-white/60 text-muted-foreground' : 'bg-burgundy-soft text-burgundy',
   110→            )}
   111→          >
   112→            {t(`choghadiya.${cur.quality}`)}
   113→          </span>
   114→          <span className="text-t12 text-muted-foreground tabular-nums">{t('home.liveEndsIn', { time: remStr })}</span>
   115→        </div>
   116→      )}
   117→      {cur && (
   118→        <div className="mt-2 h-1 rounded-full bg-gold/15 overflow-hidden" role="progressbar" aria-valuenow={Math.round(pct)} aria-valuemin={0} aria-valuemax={100}>
   119→          <div className="h-full gold-grad rounded-full" style={{ width: `${pct}%`, transition: 'width 1s linear' }} />
   120→        </div>
   121→      )}
   122→      {rahuActive && rahuEnd && (
   123→        <p className={cn('inline-flex items-center gap-1.5 rounded-full px-2.5 py-1 text-t12 font-semibold bg-burgundy-soft text-burgundy border border-burgundy/25', cur && 'mt-2')}>
   124→          <Icon name="warning" size={13} />
   125→          {t('home.liveRahu', { time: fmtT(rahuKaal?.endISO as string) })}
   126→        </p>
   127→      )}
   128→    </div>
   129→  );
   130→}

   148→/** localStorage key holding ids of sky notifications the user has opened. */
   149→const SKY_SEEN_KEY = 'ss-sky-seen-v1';
   150→
   151→function readSkySeen(): Set<string> {
   152→  if (typeof window === 'undefined') return new Set();
   153→  try {
   154→    const raw = window.localStorage.getItem(SKY_SEEN_KEY);
   155→    const arr = raw ? (JSON.parse(raw) as string[]) : [];
   156→    return new Set(Array.isArray(arr) ? arr : []);
   157→  } catch {
   158→    return new Set();
   159→  }
   160→}
   161→
   162→function persistSkySeen(ids: Set<string>) {
   163→  try {
   164→    // keep the newest 60 entries only
   165→    const arr = [...ids].slice(-60);
   166→    window.localStorage.setItem(SKY_SEEN_KEY, JSON.stringify(arr));
   167→  } catch {
   168→    /* private mode — unread state simply won't persist */
   169→  }
   170→}
   171→
   172→/** Stable unread id for a sky notification (date + kind + en name). */
   173→function skyNotifId(item: SkyItem): string {
   174→  return `${item.date}|${item.kind}|${item.name.en}`;
   175→}
