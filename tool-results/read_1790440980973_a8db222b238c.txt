    40→    40→function phaseKeyOf(p: number): string {
    41→    41→  const b = [
    42→    42→    [0, 'phase.new'], [0.0625, 'phase.waxingCrescent'], [0.1875, 'phase.firstQuarter'],
    43→    43→    [0.3125, 'phase.waxingGibbous'], [0.4375, 'phase.full'], [0.5625, 'phase.waningGibbous'],
    44→    44→    [0.6875, 'phase.lastQuarter'], [0.8125, 'phase.waningCrescent'],
    45→    45→  ] as [number, string][];
    46→    46→  for (let i = b.length - 1; i >= 0; i--) {
    47→    47→    if (p >= b[i][0]) return b[i][1];
    48→    48→  }
    49→    49→  return 'phase.new';
    50→    50→}
