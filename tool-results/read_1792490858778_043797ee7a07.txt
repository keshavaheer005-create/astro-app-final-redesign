    39→const LEGACY_CONVERSATION_ID = 'legacy';
    40→// Kept in module memory (not persisted). Navigating between tabs keeps the
    41→// current conversation, while a full app/WebView restart naturally creates a
    42→// fresh chat the next time Acharya opens.
    43→const sessionConversationIds = new Map<string, string>();
    44→// New ids have no server transcript yet, so opening one should show the welcome
    45→// state immediately without issuing an empty-history network request.

    48→function makeConversationId(): string {
    49→  try {
    50→    if (typeof crypto !== 'undefined' && typeof crypto.randomUUID === 'function') return crypto.randomUUID();
    51→  } catch { /* restricted WebView */ }
    52→  return `chat_${Date.now().toString(36)}_${Math.random().toString(36).slice(2, 12)}`;
    53→}
    54→
    55→function sessionKey(userId: string, chartId: string): string {
    56→  return `${userId}|${chartId}`;
    57→}
    58→
    59→function getSessionConversationId(userId: string, chartId: string): string {
    60→  const key = sessionKey(userId, chartId);
    61→  let id = sessionConversationIds.get(key);
    62→  if (!id) {
    63→    id = makeConversationId();
    64→    sessionConversationIds.set(key, id);
    65→    freshConversationIds.add(id);
    66→  }
    67→  return id;
    68→}
    69→
    70→function setSessionConversationId(userId: string, chartId: string, id: string): void {
    71→  sessionConversationIds.set(sessionKey(userId, chartId), id);
    72→}

    74→function conversationGroupId(updatedAt: string): 'today' | 'yesterday' | 'older' {
    75→  const date = new Date(updatedAt);
    76→  const now = new Date();
    77→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
    78→  const day = new Date(date.getFullYear(), date.getMonth(), date.getDate()).getTime();
    79→  if (day === today) return 'today';
    80→  if (day === today - 86_400_000) return 'yesterday';
    81→  return 'older';
    82→}
